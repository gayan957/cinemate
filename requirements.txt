# CineMate requirements
#
# REQUIRES PYTHON 3.11 OR 3.12. Not 3.13, not 3.14.
#
# Install:
#   python3.11 -m venv .venv
#   source .venv/bin/activate
#   pip install --upgrade pip
#   pip install -r requirements.txt
#   python -m spacy download en_core_web_sm
#   python check_env.py
#
# AFTER a successful install, lock the exact versions for the team:
#   pip freeze > requirements.lock.txt
#   git add requirements.lock.txt && git commit -m "Lock dependency versions"
#
# Teammates then install from the lock file instead:
#   pip install -r requirements.lock.txt
#
# This file uses ranges so pip can resolve a working combination.
# The lock file records exactly what it chose, so all four machines match.
